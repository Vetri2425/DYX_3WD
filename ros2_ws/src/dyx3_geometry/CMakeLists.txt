cmake_minimum_required(VERSION 3.16)
project(dyx3_geometry CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# This package has NO ROS dependency by design. Most of the rover's accuracy
# lives here, and it must be testable on a laptop in seconds rather than on the
# rover in an afternoon.
#
#   Native (no ROS):  cmake -S . -B build -DDYX3_NATIVE_TESTS=ON && ctest --test-dir build
#   In-workspace:     colcon build
option(DYX3_NATIVE_TESTS "Build standalone without ament/ROS" OFF)

add_library(dyx3_geometry STATIC
  src/angle_wrap.cpp
  src/curvature.cpp
  src/distance.cpp
  src/heading_delta.cpp
  src/line_intersection.cpp
  src/path_length.cpp
  src/perpendicular_distance.cpp
  src/project_onto_path.cpp
  src/project_onto_segment.cpp
  src/resample.cpp
  src/segment_heading.cpp
)
target_include_directories(dyx3_geometry PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
# Static, but linkable into shared libraries of other packages.
set_target_properties(dyx3_geometry PROPERTIES POSITION_INDEPENDENT_CODE ON)
# No -ffast-math, no fused multiply-add: GATE 3 demands bit-for-bit-reproducible arithmetic
# against the Python ancestors (CPython does not contract a*b+c either).
target_compile_options(dyx3_geometry PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)

function(dyx3_geometry_tests)
  foreach(t geometry_unit_test gate3_equivalence_test geometry_bag_replay_test)
    add_executable(${t} test/${t}.cpp)
    target_link_libraries(${t} dyx3_geometry)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endforeach()
  add_test(NAME geometry_unit_test COMMAND geometry_unit_test)
  # Vectors are generated by the VERBATIM Python ancestors (tools/gate3/gen_geometry_vectors.py).
  add_test(NAME gate3_equivalence_test
           COMMAND gate3_equivalence_test ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures)
  # Evidence replay against recorded bag data: SKIPPED (exit 77) until the fixture exists.
  add_test(NAME geometry_bag_replay_test
           COMMAND geometry_bag_replay_test ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures)
  set_tests_properties(geometry_bag_replay_test PROPERTIES SKIP_RETURN_CODE 77)
endfunction()

if(DYX3_NATIVE_TESTS)
  enable_testing()
  dyx3_geometry_tests()
  return()
endif()

find_package(ament_cmake REQUIRED)

install(DIRECTORY include/ DESTINATION include)
install(TARGETS dyx3_geometry EXPORT export_dyx3_geometry
        ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
ament_export_include_directories(include)
ament_export_targets(export_dyx3_geometry HAS_LIBRARY_TARGET)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)  # kept so the standard ament test deps resolve
  dyx3_geometry_tests()
endif()

ament_package()
